cobs_encode_post_zero_pad_vargs(  dst,  dst_length,  valist )
 
    end = dst + dst_length;

    uint8_t code, *code_ptr;  

    //StartBlock();
        code_ptr = dst++;
        code = 1;

    while(true)
     
        // 取得当前列表中参数元素的长度
        arg_len = va_arg(valist, int32_t);

        if(arg_len == 0) //碰到args，跳出while
            break;
 
        //取得当前列表中参数元素的数据指针
        arg_data = va_arg(valist, uint8_t *);
 
        for( b = 0; b < arg_len; b++)
        
            if (code != 0xFF)
            
                c = *arg_data++;

                if (c != 0)
                
                    *dst++ = c;
                    code++;
                    continue;     //绕过下面的 FinishBlock 和 StartBlock
                
             
            //FinishBlock();      
               *code_ptr = code

            //StartBlock();
               code_ptr = dst++;
               code = 1;
     

    //FinishBlock();
        *code_ptr = code


    //Make sure we didn't overrun the buffer
    ASSERT(dst <= end);

    //最后加个0做结尾符
    while(dst < end) *dst++ = 0;
